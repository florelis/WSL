foreach(variant IN ITEMS fake wrong_abi missing_exports)
    add_library(compose_${variant} SHARED FakeLibrary.cpp)
    target_link_libraries(compose_${variant} PRIVATE compose_parser)
    target_include_directories(compose_${variant} PRIVATE "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/$<CONFIG>")
    target_precompile_headers(compose_${variant} PRIVATE precomp.h)
    target_link_options(compose_${variant} PRIVATE /EXPORT:ComposeAbiVersion /EXPORT:OutstandingAllocations)
    if(NOT variant STREQUAL "missing_exports")
        target_link_options(compose_${variant} PRIVATE /EXPORT:ComposeNormalize /EXPORT:ComposeFree)
    endif()
    set_target_properties(compose_${variant} PROPERTIES FOLDER tests)
    add_dependencies(wsltests compose_${variant})
endforeach()
target_compile_definitions(compose_wrong_abi PRIVATE WRONG_ABI)
target_compile_definitions(compose_missing_exports PRIVATE MISSING_EXPORTS)
target_include_directories(wsltests PRIVATE "${CMAKE_RUNTIME_OUTPUT_DIRECTORY}/$<CONFIG>")
